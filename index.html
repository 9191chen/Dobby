Option Explicit Sub 發信草稿_依設定產生() Dim olApp As Object, olMail As Object Dim ws As Worksheet Set ws = ThisWorkbook.Sheets("test") Dim lastRow As Long, i As Long Dim groupDict As Object Set groupDict = CreateObject("Scripting.Dictionary") Dim doctorName As String, salesName As String, clinicName As String Dim email As String, ccEmail As String, groupKey As String Dim k As Variant, firstRow As Long Dim groupItems As Collection Dim totalAmt As Double Dim createdCount As Long, skippedCount As Long Dim groupContentHtml As String Dim folderPath As String folderPath = """C:\Users\joychen1\OneDrive - Align Technology, Inc\桌面\自動化發信機器人\測試用對帳單""" Set olApp = CreateObject("Outlook.Application") If Dir(folderPath, vbDirectory) = "" Then MsgBox "找不到 PDF 資料夾：" & vbCrLf & folderPath, vbExclamation Exit Sub End If lastRow = ws.Cells(ws.Rows.Count, "D").End(xlUp).Row '== 分組：醫師(D) + 業務(F) == For i = 2 To lastRow doctorName = Trim(CStr(ws.Cells(i, "D").Value)) salesName = Trim(CStr(ws.Cells(i, "F").Value)) If doctorName <> "" And salesName <> "" Then groupKey = doctorName & "||" & salesName If Not groupDict.Exists(groupKey) Then Set groupDict(groupKey) = New Collection End If groupDict(groupKey).Add i End If Next i '== 建立草稿 == For Each k In groupDict.Keys Set groupItems = groupDict(k) firstRow = groupItems(1) doctorName = Trim(CStr(ws.Cells(firstRow, "D").Value)) salesName = Trim(CStr(ws.Cells(firstRow, "F").Value)) clinicName = Trim(CStr(ws.Cells(firstRow, "C").Value)) email = Trim(CStr(ws.Cells(firstRow, "A").Value)) ccEmail = Trim(CStr(ws.Cells(firstRow, "G").Value)) totalAmt = GetGroupTotal(ws, groupItems) If True Then ' 無金額門檻，全部發信 groupContentHtml = GetGroupContentHtml(ws, groupItems, totalAmt) Set olMail = olApp.CreateItem(0) With olMail .To = email If ccEmail <> "" Then .CC = ccEmail .Subject = "逾期款提醒通知" .HTMLBody = "
" & _ """ & doctorName & ""醫師您好，

" & _ groupContentHtml & "

" & _ "您目前尚有逾期款 "" & Format(totalAmt, ""#,##0"") & ""。
" & _ "請於22前完成繳款，以避免您的帳號權限受到影響。
" & _ "若您已完成繳款，請提供：1. 日期 2.金額 3. 匯款條 or (含)帳戶成功扣款的畫面截圖
" & _ "
" & _ "若款項有任何問題，請再回信與我們聯繫，謝謝！

" & _ "請於本月22號前完成繳款。
" & _ "Best regards,
" & _ "Finance
" & _ "
" Call AttachPdfFilesForGroup(ws, groupItems, olMail, folderPath) .Save End With createdCount = createdCount + 1 Else skippedCount = skippedCount + 1 End If Next k MsgBox "完成！" & vbCrLf & _ "已建立草稿：" & createdCount & " 封" & vbCrLf & _ "跳過：" & skippedCount & " 組", vbInformation End Sub Function GetGroupTotal(ws As Worksheet, groupItems As Collection) As Double Dim i As Long, r As Long, total As Double For i = 1 To groupItems.Count r = groupItems(i) If IsNumeric(ws.Cells(r, "E").Value) Then total = total + CDbl(ws.Cells(r, "E").Value) End If Next i GetGroupTotal = total End Function Function GetGroupContentHtml(ws As Worksheet, groupItems As Collection, totalAmt As Double) As String If groupItems.Count = 1 Then GetGroupContentHtml = "您目前尚欠的逾期款為 " & _ "$" & _ Format(totalAmt, "#,##0") & "" Else GetGroupContentHtml = "下列是您目前尚欠的逾期款明細：

" & _ BuildHtmlTable(ws, groupItems) End If End Function Function BuildHtmlTable(ws As Worksheet, groupItems As Collection) As String Dim html As String Dim i As Long, r As Long, totalAmt As Double html = "" & _ "" & _ "" & _ "" totalAmt = 0 For i = 1 To groupItems.Count r = groupItems(i) html = html & "" & _ "" & _ "" & _ "" & _ "" & _ "" If IsNumeric(ws.Cells(r, "E").Value) Then totalAmt = totalAmt + CDbl(ws.Cells(r, "E").Value) End If Next i html = html & "" & _ "" & _ "" & _ "
Account診所醫師名逾期款
" & ws.Cells(r, "B").Value & "" & ws.Cells(r, "C").Value & "" & ws.Cells(r, "D").Value & "$" & Format(ws.Cells(r, "E").Value, "#,##0") & "
加總$" & Format(totalAmt, "#,##0") & "
" BuildHtmlTable = html End Function Sub AttachPdfFilesForGroup(ws As Worksheet, groupItems As Collection, olMail As Object, folderPath As String) Dim attachedDict As Object Set attachedDict = CreateObject("Scripting.Dictionary") Dim allFiles() As String Dim fileCount As Long, f As String fileCount = 0 f = Dir(folderPath & "*.pdf") Do While f <> "" ReDim Preserve allFiles(fileCount) allFiles(fileCount) = f fileCount = fileCount + 1 f = Dir() Loop Dim i As Long, j As Long, r As Long Dim accountValue As String, fullPath As String For i = 1 To groupItems.Count r = groupItems(i) accountValue = Trim(CStr(ws.Cells(r, "B").Value)) If accountValue <> "" And fileCount > 0 Then For j = 0 To fileCount - 1 If IsPdfMatch(allFiles(j), accountValue) Then fullPath = folderPath & "" & allFiles(j) If Not attachedDict.Exists(fullPath) Then olMail.Attachments.Add fullPath attachedDict.Add fullPath, True End If End If Next j End If Next i End Sub Function IsPdfMatch(fileName As String, accountValue As String) As Boolean Dim fd As String, ad As String fd = RemoveLeadingZeros(DigitsOnly(LCase$(fileName))) ad = RemoveLeadingZeros(DigitsOnly(LCase$(Trim$(CStr(accountValue))))) If ad <> "" And fd <> "" Then IsPdfMatch = (fd = ad) Else IsPdfMatch = False End If End Function Function DigitsOnly(ByVal txt As String) As String Dim i As Long, ch As String, result As String For i = 1 To Len(txt) ch = Mid$(txt, i, 1) If ch Like "#" Then result = result & ch Next i DigitsOnly = result End Function Function RemoveLeadingZeros(ByVal txt As String) As String Do While Len(txt) > 1 And Left$(txt, 1) = "0" txt = Mid$(txt, 2) Loop RemoveLeadingZeros = txt End Function
